# Libraries for running the training notebook (football-transfer-prediction.ipynb)
# on your own computer instead of Kaggle.
#
# The web app does NOT need this file. It installs only requirements.txt.
#
# How to use (Python 3.12 recommended, the same version Kaggle used):
#   python -m venv .venv-train
#   source .venv-train/bin/activate        # Windows: .venv-train\Scripts\activate
#   pip install -r requirements-train.txt
#   jupyter notebook football-transfer-prediction.ipynb
#
# Before running the notebook:
#   1. Download the dataset "Football Data from Transfermarkt" from
#      https://www.kaggle.com/datasets/davidcariboo/player-scores
#   2. The notebook reads files from /kaggle/input/player-scores/. Change those
#      paths to wherever you saved players.csv, player_valuations.csv and
#      appearances.csv.
#
# The dataset is updated regularly, so your results will differ slightly from
# the numbers in the README (May 2026 run).

pandas
numpy
scikit-learn

# Keep this the same version as in requirements.txt. A model saved by a newer
# XGBoost may not load correctly in an older one.
xgboost==3.1.3

notebook
